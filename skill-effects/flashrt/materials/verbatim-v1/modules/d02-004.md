**Parameterized ops — subclass `IROperator`.** Use this when the
same kind of operator is instantiated many times with different
arguments (e.g., one denoising step per inference step, each with
its own KV cache):

```python
from flashrt.ir import IROperator, OpType, TensorPort

class DiTDenoiseStep(IROperator):
    def __init__(self, step_idx: int, dit_runner):
        cache = f"step_{step_idx}_kv"
        super().__init__(
            name=f"dit_denoise_{step_idx}",
            op_type=OpType.EXPOSED,
            inputs=[TensorPort("latents"), TensorPort("timestep")],
            outputs=[TensorPort("noise_pred")],
            state_reads=[cache],
            state_writes=[cache],  # declares in-place mutation
        )
        self.cache = cache
        self.dit_runner = dit_runner

    def execute(self, inputs, ctx, state):
        kv = state.get(self.cache)          # ref to the cache object
        self.dit_runner.kv_memory = kv       # runner mutates it in place
        noise_pred = self.dit_runner.run(
            latents=inputs["latents"],
            timestep=inputs["timestep"],
            ...,
        )
        return {"noise_pred": noise_pred}
```
